from pathlib import Path import base64 from PIL import Image html_path = Path("/mnt/data/ruta_exito_comercial_v2.html") fed_src = Path("/mnt/data/logo-fedecamaras(1).png") ccl_src = Path("/mnt/data/CCL(1).jpg") # El archivo de Fedecámaras tiene extensión .png, pero internamente es JPEG. # Lo convertimos a PNG real para evitar problemas de compatibilidad en navegadores/GitHub Pages. fed_png = Path("/mnt/data/logo-fedecamaras-real.png") Image.open(fed_src).convert("RGBA").save(fed_png, "PNG") def data_uri(path, mime): return f"data:{mime};base64," + base64.b64encode(path.read_bytes()).decode("ascii") fed_uri = data_uri(fed_png, "image/png") ccl_uri = data_uri(ccl_src, "image/jpeg") html = html_path.read_text(encoding="utf-8") # Reemplaza cualquier URI anterior de los logos, manteniendo el HTML autónomo. import re html = re.sub(r'data:image/(?:png|jpeg);base64,[A-Za-z0-9+/=]+', lambda m: fed_uri if 'logo-fed' in html[max(0,m.start()-150):m.start()] else m.group(0), html) # Para hacerlo inequívoco, reconstruimos los src de los logos según sus alt. html = re.sub( r'(]+alt="Fedecámaras Anzoátegui"[^>]+src=")data:image/(?:png|jpeg);base64,[^"]+(")', r'\1' + fed_uri + r'\2', html ) html = re.sub( r'(]+alt="Cámara de Comercio Lechería"[^>]+src=")data:image/(?:png|jpeg);base64,[^"]+(")', r'\1' + ccl_uri + r'\2', html ) # También corrige el orden si el src aparece antes del alt. html = re.sub( r'(]+src=")data:image/(?:png|jpeg);base64,[^"]+("[^>]+alt="Fedecámaras Anzoátegui")', r'\1' + fed_uri + r'\2', html ) html = re.sub( r'(]+src=")data:image/(?:png|jpeg);base64,[^"]+("[^>]+alt="Cámara de Comercio Lechería")', r'\1' + ccl_uri + r'\2', html ) # Si el reemplazo anterior no encontró los tags por formato, genera una versión # robusta sustituyendo las URIs por posición: Fedecámaras primero, CCL segundo. matches = list(re.finditer(r'data:image/(?:png|jpeg);base64,[A-Za-z0-9+/=]+', html)) if len(matches) >= 2: # Los primeros usos corresponden a Fedecámaras y CCL. replacements = {matches[0].span(): fed_uri, matches[1].span(): ccl_uri} for span, uri in sorted(replacements.items(), reverse=True): html = html[:span[0]] + uri + html[span[1]:] fixed = Path("/mnt/data/ruta_exito_comercial_v3_imagenes.html") fixed.write_text(html, encoding="utf-8") print(f"Versión corregida: {fixed}") print(f"Tamaño: {fixed.stat().st_size/1024:.1f} KB") print("Fedecámaras convertido a PNG real y ambos logos incrustados en el HTML.")
